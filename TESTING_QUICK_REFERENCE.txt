COMMON ASSERTIONS
─────────────────────────────────────────────────────────────────────────────

Equality:
  assert_equal(expected, actual)
  assert_nil(value)
  assert_empty(array)

HTTP Response:
  assert_response :success         # 200
  assert_response :created         # 201
  assert_response :redirect        # 3xx
  assert_response :unprocessable_entity  # 422

Redirect:
  assert_redirected_to root_path
  assert_redirected_to user_path(@user)

HTML/Content:
  assert_select "div"
  assert_select "div#id"
  assert_select "div", text: "content"
  assert_match(/pattern/, @response.body)

Database:
  assert_difference('User.count', 1) { post users_path }
  assert_no_difference('User.count') { get users_path }

Exceptions:
  assert_raises(StandardError) { code }

USEFUL PATTERNS
─────────────────────────────────────────────────────────────────────────────

Test both success and failure:
  test "valid case" do
    result = method(valid_params)
    assert result
  end
  
  test "invalid case" do
    result = method(invalid_params)
    assert_nil result
  end

Test scoping (user isolation):
  test "respects user isolation" do
    other_user = users(:two)
    @user.create_record(name: "test")
    
    results = MyService.search("test", @user)
    results_other = MyService.search("test", other_user)
    
    assert_equal 1, results.count
    assert_equal 1, results_other.count  # Should only see own
  end

Test idempotency:
  test "prevents duplicates" do
    create_notification(...)
    create_notification(...)
    
    assert_equal 1, Notification.count
  end
